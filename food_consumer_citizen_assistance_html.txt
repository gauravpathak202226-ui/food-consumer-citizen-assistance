function show(id){

/* ---------------- VOICE ---------------- */

function setupVoice(){

 const SR =
   window.SpeechRecognition ||
   window.webkitSpeechRecognition;

 if(!SR){
   recognition = null;
   return false;
 }

 recognition = new SR();
 recognition.continuous = false;
 recognition.interimResults = false;

 return true;
}

function listen(callback,status){

 if(!recognition && !setupVoice()){

   status.textContent =
   "Voice recognition is not supported in this browser. Please type.";

   return;
 }

 recognition.lang =
   language==="hi" ? "hi-IN" : "en-IN";

 status.textContent =
   language==="hi"
   ? "सुन रहा है... बोलिए..."
   : "Listening... speak now";

 try{
   recognition.start();
 }catch(e){}

 recognition.onresult=function(event){

   const text =
     event.results[0][0].transcript;

   callback(text);

   status.textContent =
     language==="hi"
     ? "आवाज़ प्राप्त हुई"
     : "Voice captured";
 };

 recognition.onerror=function(event){

   status.textContent =
     "Voice error: "+event.error;
 };
}

function speak(text){

 if(!window.speechSynthesis) return;

 speechSynthesis.cancel();

 const u =
   new SpeechSynthesisUtterance(text);

 u.lang =
   language==="hi" ? "hi-IN" : "en-IN";

 u.rate=.9;

 speechSynthesis.speak(u);
}


/* ---------------- FLOW DETECTION ---------------- */

function detectFlow(text){

 const s=text.toLowerCase();

 for(const [key,flow] of Object.entries(FLOWS)){

   for(const keyword of flow.keywords){

     if(s.includes(keyword.toLowerCase()))
       return key;
   }
 }

 return null;
}

     input.value=state.answers[q[0]];

   area.appendChild(input);
   input.focus();

   language==="hi"
   ? "उत्तर भरकर 'सही है, आगे बढ़ें' दबाएं।"
   : "Enter the answer and press Confirm & Continue.";

}

function saveCurrent(){

 const q =
   state.questions[state.index];

 let value="";

 if(q[3]==="yesno"){

   value =
     document.getElementById("answerArea")
     .dataset.value || "";

 }else{

   value =
     document.getElementById("answer")
     .value.trim();
 }

 if(!value){

   alert(
     language==="hi"
     ? "कृपया उत्तर दें।"
     : "Please provide an answer."
   );

   return;
 }

 if(q[0]==="mobile" &&
    !/^[0-9]{10}$/.test(value)){

   alert("Please enter a valid 10 digit mobile number.");
   return;
 }

 if(q[0]==="aadhaar" &&
    !/^[0-9]{12}$/.test(value.replace(/\s/g,""))){

   alert("Please enter a valid 12 digit Aadhaar number.");
   return;
 }

 const confirmed =
   confirm(
    (language==="hi"
     ? "आपने कहा: "
     : "You entered: ")
    +value+
    "\n\n"+
    (language==="hi"
     ? "क्या यह सही है?"
     : "Is this correct?")
   );

 if(!confirmed) return;

 state.answers[q[0]]=value;

 /*
   Guided classification:
   hasRation = yes
   receivedThisMonth = no
   => PDS not received
   fullQuantity = no
   => less grain
 */
 if(!state.flow && state.index===2){

   if(state.answers.receivedThisMonth==="no"){

     state.flow="pds_not_received";

   }else if(state.answers.fullQuantity==="no"){

     state.flow="less_grain";

   }else{

     state.flow="ration_problem";
   }

   /*
     Continue with common details.
   */
   state.questions=[
     ...COMMON,
     ...(FLOWS[state.flow].questions||[])
   ];

   state.index=0;

   renderQuestion();
   return;
 }

 state.index++;

 if(state.index < state.questions.length){

   renderQuestion();

 }else{

   showReview();
 }
}


/* ---------------- VOICE ANSWER ---------------- */

function voiceAnswer(){

 listen(
   text=>{

     const q=state.questions[state.index];

     if(q[3]==="yesno"){

       const area=
         document.getElementById("answerArea");

       const s=text.toLowerCase();

       if(
        s.includes("हाँ") ||
        s.includes("हां") ||
        s.includes("yes")
       ){

         area.dataset.value="yes";

       }else if(
        s.includes("नहीं") ||
        s.includes("नही") ||
        s.includes("no")
       ){

         area.dataset.value="no";

       }else{

         alert(
          "कृपया हाँ या नहीं बोलें।"
         );
         return;
       }

     }else{

       document.getElementById("answer").value=text;
     }

     speak(
       language==="hi"
       ? "आपने कहा: "+text
       : "You said: "+text
     );

   },

   document.getElementById("questionStatus")
 );
}


/* ---------------- REVIEW ---------------- */

function showReview(){

 show("review");

 let html="";

 state.questions.forEach(q=>{

   const value =
     state.answers[q[0]] || "";

   html+=
    `<div class="review-item">
       <div class="review-key">
         ${esc(q[language==="hi"?1:2])}
       </div>
       <div class="review-value">
         ${esc(value)}
       </div>
     </div>`;
 });

 document.getElementById("reviewContent")
 .innerHTML=html;
}

function generate(){

 show("application");

 .textContent =
 new Date().toLocaleDateString("en-IN");

 document.getElementById("paperTitle")
 .textContent =
 FLOWS[state.flow]?.title ||

 let html=

 COMMON.forEach(q=>{
   html+=paperRow(
     q[1],
     state.answers[q[0]]
   );
 });

 html+="</div>";

 html+=
 `<div class="paper-section">
    <h3>Application / Grievance Details</h3>`;

 (FLOWS[state.flow]?.questions||[])
 .forEach(q=>{
   html+=paperRow(
     q[1],
     state.answers[q[0]]
   );
 });

 html+="</div>";

 html+=
 `<div class="paper-section">
    <h3>Request</h3>
    <p>
      I request the concerned authority to kindly
      examine the above matter and take necessary
      action as per applicable rules.
    </p>
  </div>`;

 document.getElementById("paperBody")
 .innerHTML=html;
}


function paperRow(label,value){

 return `
 <div class="paper-row">
   <strong>${esc(label)}</strong>
   <span>${esc(value)}</span>
 </div>`;
}


/* ---------------- SECURITY ---------------- */

function esc(value){

 return String(value ?? "")
 .replace(/[&<>"']/g,
 c=>({
  "&":"&amp;",
  "<":"&lt;",
  ">":"&gt;",
  '"':"&quot;",
  "'":"&#039;"
 }[c]));
}
